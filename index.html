function doPost(e) { try { // Membaca parameter yang dikirim lewat metode URLSearchParams const requestData = e.parameter; const action = requestData.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === "login") { const sheet = ss.getSheetByName("Akun"); if (!sheet) return jsonResponse({ status: "error", message: "Sheet 'Akun' tidak ditemukan!" }); const data = sheet.getDataRange().getValues(); const u = requestData.username ? requestData.username.toString().trim() : ""; const p = requestData.password ? requestData.password.toString().trim() : ""; if (!u || !p) { return jsonResponse({ status: "error", message: "Username dan password wajib diisi!" }); } for (let i = 1; i < data.length; i++) { if (!data[i]) continue; const dbNpsn = data[i][0].toString().trim(); const dbPass = data[i][1].toString().trim(); const dbSekolah = data[i][2].toString().trim(); const dbRole = data[i][3].toString().trim(); if (u === dbNpsn && p === dbPass) { return jsonResponse({ status: "success", user: { sekolah: dbSekolah, role: dbRole, kecamatan: "Kab. Rembang", namaTim: "HARYANTO" } }); } } return jsonResponse({ status: "error", message: "NPSN atau password salah!" }); } if (action === "insertProgres") { const sheet = ss.getSheetByName("Database"); if (!sheet) return jsonResponse({ status: "error", message: "Sheet 'Database' tidak ditemukan!" }); const idInput = "PRG-" + Math.floor(1000 + Math.random() * 9000); sheet.appendRow([ idInput, requestData.npsn || "", requestData.namaSd || "", requestData.kecamatan || "", requestData.namaMenu || "", Number(requestData.nilai) || 0, Number(requestData.volumeRuang) || 0, requestData.namaTim || "" ]); return jsonResponse({ status: "success", message: "Data progres berhasil disimpan." }); } return jsonResponse({ status: "error", message: "Aksi POST tidak valid." }); } catch (error) { return jsonResponse({ status: "error", message: "Error sistem: " + error.toString() }); } }
