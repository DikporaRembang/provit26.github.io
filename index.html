function doGet(e) { var action = e.parameter.action; if (action === "manageMasterSd") { return manageMasterSd(e); } // ... (action lain tetap ada) } function manageMasterSd(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Master_SD"); // Pastikan nama sheet sesuai if (!sheet) { return responseJSON({ status: "error", message: "Sheet Master_SD tidak ditemukan" }); } var subAction = e.parameter.subAction; // 'add', 'edit', atau 'delete' var npsn = String(e.parameter.npsn || "").trim(); var password = e.parameter.password || ""; var sekolah = e.parameter.sekolah || ""; var kecamatan = e.parameter.kecamatan || ""; var masaPks = e.parameter.masaPks || ""; var mulai = e.parameter.mulai || ""; var selesai = e.parameter.selesai || ""; var role = e.parameter.role || "SEKOLAH"; var data = sheet.getDataRange().getValues(); // 1. TAMBAH DATA BARU (ADD) if (subAction === "add") { // Cek apakah NPSN sudah ada for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === npsn) { return responseJSON({ status: "error", message: "NPSN sudah terdaftar!" }); } } // Baris kolom: NPSN | Password | Nama SD | Kecamatan | Masa PKS | Mulai | Selesai | Role sheet.appendRow([npsn, password, sekolah, kecamatan, masaPks, mulai, selesai, role]); return responseJSON({ status: "success", message: "Data SD berhasil ditambahkan!" }); } // 2. PERBARUI DATA (EDIT) else if (subAction === "edit") { var rowIndex = -1; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === npsn) { rowIndex = i + 1; // +1 karena baris Google Sheet dimulai dari index 1 break; } } if (rowIndex !== -1) { // Update baris yang ditemukan sesuai urutan kolom sheet.getRange(rowIndex, 1, 1, 8).setValues([[ npsn, password, sekolah, kecamatan, masaPks, mulai, selesai, role ]]); return responseJSON({ status: "success", message: "Data SD berhasil diperbarui!" }); } else { return responseJSON({ status: "error", message: "Data NPSN tidak ditemukan!" }); } } // 3. HAPUS DATA (DELETE) else if (subAction === "delete") { for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === npsn) { sheet.deleteRow(i + 1); return responseJSON({ status: "success", message: "Data SD berhasil dihapus!" }); } } return responseJSON({ status: "error", message: "Data NPSN tidak ditemukan untuk dihapus!" }); } return responseJSON({ status: "error", message: "Aksi tidak valid" }); } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
